 * Problem: Remove Duplicates from Sorted Array (#26)

 * Solved on: 2026-10-02
 * Runtime: 0 ms | Memory: 22.8 MB
 */

class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int j = 0;
        for(int i = 1; i < nums.size(); i++){
            if(nums[i] != nums[j]){
                j++;
                nums[j] = nums[i];
            }
        }
        return ++j;
    }
};
