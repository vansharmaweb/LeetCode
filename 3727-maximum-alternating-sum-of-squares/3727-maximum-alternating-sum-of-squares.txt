public:
    long long maxAlternatingSum(vector<int>& nums) {
        for(int i = 0; i < nums.size(); i++){
            nums[i] = pow(nums[i], 2);
        }
        sort(nums.begin(), nums.end());
        return nums.size() % 2 == 0 ? sum : sum += nums[nums.size()/2];
        int i = 0; 
        int j = nums.size() - 1;
        while(i < j){
            sum += nums[j];
        }
            sum -= nums[i];
            j--;
            i++;
    }
        long long sum = 0;
