class Solution {
public:
    int numPairsDivisibleBy60(vector<int>& time) {
        unordered_map<int, int> mp;
        int count = 0;
        for(int val : time){
            int r = val % 60;
            int need = (60 - r) % 60;
                count += mp[need];
            mp[r]++;
        }
        return count;
    }
};
