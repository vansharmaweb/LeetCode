class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        int i = 0; int j = nums.size() - 1;
        int k = nums.size() - 1;
        vector<int> ans(nums.size());
        while(i <= j){
            int t1 = abs(nums[i]);
            int t2 = abs(nums[j]);
            if(t1 > t2){
                ans[k] = nums[i] * nums[i];
                i++;
            }else{
                ans[k] = nums[j] * nums[j];
                j--;
            }
            k--;
