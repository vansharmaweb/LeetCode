public:
    int maxArea(vector<int>& height) {
        int lp = 0, rp = height.size()-1, ans = 0;
        while(lp < rp){
            int w = rp - lp;
            int h = min(height[lp], height[rp]);
            int curr_wat = w * h;
            ans = max(curr_wat, ans);
            if(height[lp] < height[rp])
             lp++;
            else 
             rp--;
        }
        return ans;
    }
};
