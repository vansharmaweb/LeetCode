class Solution {
public:
    bool checkPerfectNumber(int num) {
        int count = 0;
        int x = num;
        for(int i = 1; i <= sqrt(num); i++){
            if(num % i == 0){
                count += i;
            }
        }
        return count == x;
    }
};
