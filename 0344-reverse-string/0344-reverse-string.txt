 * Problem: Reverse String (#344)

 * Solved on: 2026-10-02

class Solution {
public:
    void reverseString(vector<char>& s) {
        int i = 0, j = s.size() - 1;

        while(i < j){
            swap(s[i], s[j]);
            i++;
            j--;
        }
    }
};
