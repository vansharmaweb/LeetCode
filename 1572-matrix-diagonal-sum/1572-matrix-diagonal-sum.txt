 * Solved on: 2026-10-02

class Solution {
public:
    int diagonalSum(vector<vector<int>>& mat) {
        int d1 = 0;
        int d2 = 0;
        for(int i = 0; i < mat.size(); i++){
            for(int j = 0;j < mat[0].size(); j++){
                if(i == j){
                    d1 += mat[i][j];
                }else if(i + j == mat.size() - 1){
                    d2 += mat[i][j];
                }
            }
        }
        return d1 + d2;
    }
};
