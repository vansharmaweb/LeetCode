/*
 * Problem: Best Time to Buy and Sell Stock (#121)
 * Difficulty: Easy
 * Topics: Array, Dynamic Programming
 * Language: Sort by:
 * LeetCode URL: https://leetcode.com/problems/best-time-to-buy-and-sell-stock/
 * Solved on: 2026-10-02
 * Runtime: 7 ms | Memory: 97.3 MB
 */

class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int maxprof = 0, bestbuy = prices[0];
        for(int i = 1; i < prices.size(); i++){
            if(prices[i] >= bestbuy){
                maxprof = max(maxprof, prices[i] - bestbuy);
            }
            bestbuy = min(bestbuy, prices[i]);
        }
        return maxprof;
    }
};
