    int absDifference(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end());
        int i = 0;
        int j = nums.size()-1;
        int s1 = 0;
        int s2 = 0;
        for(int i = 0; i < k; i++){
            s1 += nums[i];
            s2 += nums[j];
            j--;
        }
        return abs(s1 - s2);
    }
};
