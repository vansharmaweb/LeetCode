 * Problem: Reverse Substrings Between Each Pair of Parentheses (#1190)

class Solution {
public:
    string reverseParentheses(string s) {
        string st;
        stack<int> stk;
        for(char c : s){
            if(c == '('){
                stk.push(st.size());
            }else if(c == ')'){
                int start = stk.top();
                stk.pop();
                reverse(st.begin() + start, st.end());
            }else{
                st += c;
            }
        }
        return st;
