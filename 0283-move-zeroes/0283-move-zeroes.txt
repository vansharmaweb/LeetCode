 * Problem: Move Zeroes (#283)

 * Solved on: 2026-10-02

class Solution {
public:
    void moveZeroes(vector<int>& nums) {
        int i = 0, j = 1;
        while(j <= nums.size() - 1){
            if(nums[i] == 0 && nums[j] != 0){
                swap(nums[i], nums[j]);
                i++;
            }
            else if(nums[i] != 0){
                i++;
            }
            j++;
        }
    }
};
