class Solution {
public:
    int findMiddleIndex(vector<int>& nums) {
        int total = 0;
        for(int i = 0; i < nums.size(); i++){
            total = total + nums[i];
        }
        int leftSum = 0;
        for(int i = 0; i < nums.size(); i++){
            if(leftSum == total - leftSum - nums[i]) return i;
            leftSum = leftSum + nums[i]; 
        }
        return -1;
    }
};
