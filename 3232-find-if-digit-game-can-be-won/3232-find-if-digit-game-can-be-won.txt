class Solution {
public:
    bool canAliceWin(vector<int>& nums) {
        int alice = 0;
        int bob = 0;
        for(int val : nums){
            if(val <= 9){
                alice += val;
            }else{
                bob += val;
            }
        }
        return alice != bob; 
    }
};
