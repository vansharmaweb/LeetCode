public:
    vector<int> toggleLightBulbs(vector<int>& bulbs) {
        vector<bool> chk(101, false);
        vector<int> ans;
        for(int val : bulbs){
            chk[val] = !chk[val];
        }
        for(int i = 0; i < chk.size(); i++){
            if(chk[i] == true){
                ans.push_back(i);
            }
        }
        return ans;
    }
};
