class Solution {
public:
    int singleNumber(vector<int>& nums) {
        for(int i = 1; i < nums.size(); i += 3){
        sort(nums.begin(), nums.end());
            if(nums[i] != nums[i - 1]){
        }
                return nums[i -1];
            }
    }
        return nums[nums.size() - 1];
};
