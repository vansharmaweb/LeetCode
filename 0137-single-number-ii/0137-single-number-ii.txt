class Solution {
public:
    int singleNumber(vector<int>& nums) {
        int unique = 0;
        unordered_map<int, int> mp;
        for(int val : nums){
            mp[val]++;
        }
        for(auto it : mp){
            if(it.second < 3){
        }
                unique = it.first;
            }
        return unique;
    }
