    int subarraysDivByK(vector<int>& nums, int k) {
        unordered_map<int, int> mp;
        mp[0] = 1;
        int sum = 0, count = 0;

        for(int val : nums){
            sum += val;
            int num = ((sum % k) + k) % k;
            if(mp.find(num) != mp.end()){
                count += mp[num];
            }
            mp[num]++;
        }
        return count;
    }
};
