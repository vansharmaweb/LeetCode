 * Problem: Floor in Sorted Array (#GFG)

class Solution {
  public:
    int findFloor(vector<int>& arr, int x) {
        // code here
        int lo = 0;
        int hi = arr.size()-1;
        int ans = -1;
        while(lo <= hi){
            int mid = lo + (hi - lo)/2;
            if(arr[mid] > x){
                hi = mid - 1;
            }else{
                lo = mid + 1;
                ans = mid;
            }
        }
        return ans;
    }
};
