/*
 * Problem: Second Largest (#GFG)
 * Difficulty: Medium
 * Topics: Uncategorized

 * LeetCode URL: https://leetcode.com/problems/1/
 * Solved on: 2026-10-25
 */

class Solution {
  public:
    int getSecondLargest(vector<int> &arr) {
        // code here
        int largest = INT_MIN;
        for(int val : arr){
            if(val > largest){
                largest = val;
            }
        }
        int secLargest = INT_MIN;
        for(int val : arr){
            if(val == largest){
                continue;
            }else if(val > secLargest){
                secLargest = val;
            }
        }
        return secLargest > 0 ? secLargest : -1;
    }
};
