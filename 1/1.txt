 * Problem: Implement Lower Bound (#GFG)

class Solution {
  public:
    int lowerBound(vector<int>& arr, int x) {
        // code here
        int lo = 0;
        int hi = arr.size()-1;
        while(lo <= hi){
            int mid = lo + (hi - lo)/2;
            if(arr[mid] >= x){
                hi = mid - 1;
            }else{
                lo = mid + 1;
            }
        }
        return lo;
        
    }
};
