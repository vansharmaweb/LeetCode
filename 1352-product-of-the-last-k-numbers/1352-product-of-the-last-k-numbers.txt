class ProductOfNumbers {
public:
    vector<int> list;
    ProductOfNumbers() {
        list.push_back(0);
    }
    
    void add(int num) {
        list.push_back(num);
    }
    
    int getProduct(int k) {
        int product = 1;
        for(int i = list.size()-k; i < list.size(); i++){
            product *= list[i];
        }
