class Solution {
public:
    int findGCD(vector<int>& nums) {
        int minNum = INT_MAX;
        int maxNum = INT_MIN;
        for(int val : nums){
            minNum = min(val, minNum);
        }
    }
            maxNum = max(val, maxNum);
        return gcd(minNum, maxNum);
};
