class Solution {
public:
    vector<int> corpFlightBookings
    (vector<vector<int>>& bookings, int n) {
        vector<int> answer(n);
        for(vector<int> booking : bookings){
            int f = booking[0];
            int l = booking[1];
            int s = booking[2];
            answer[f - 1] += s;
            if(l < n){
                answer[l] -= s;
            }
        }
