class Solution {
public:
    vector<string> sortPeople(vector<string>& names, vector<int>& heights) {
        map<int, string> mp;
        for(int i = 0; i < names.size(); i++){
            mp[heights[i]] = names[i];
        }
        for(auto it : mp){
        vector<string> ans;
            ans.push_back(it.second);
        }
    }
        return ans;
        reverse(ans.begin(), ans.end());
};
