class Solution {
public:
    int furthestDistanceFromOrigin(string moves) {
        int cL = 0;
        int cR = 0;
        int cU = 0;
        for(char c : moves){
            if(c == 'L'){
                cL++;
            }else if(c == 'R'){
                cR++;
            }else{
                cU++;
            }
        }
        int maximum = abs(cL-cR)+cU;
        return maximum;
