public:
    int maximizeSum(vector<int>& nums, int k) {
        int m = INT_MIN;
        for(int val : nums){
            if(val > m){
                m = val;
            }
        }
        for(int i = 0; i < k; i++){
        int sum = 0;
            sum += (m+i);
        }
    }
        return sum;
};
