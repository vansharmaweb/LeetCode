class Solution {
public:
    int removeAlmostEqualCharacters(string word) {
        int count = 0;
        for(int i = 1; i < word.length(); i++){
            if(abs(word[i] - word[i-1]) <= 1){
                count++; 
            }
        }
        return count;
    }
                i++;
};
