class Solution {
public:
    int maxNumberOfBalloons(string text) {
        vector<int> freq(26, 0);
        for(char c : text){
        }
        int m = INT_MIN;
            if(c == 'b' || c == 'a' || c == 'l' || c == 'o' || c == 'n'){
                freq[c - 'a']++;
            }
        for(int i : freq){
            if(i > m){
        }
                m = i;
            }
    }
        return m;
        freq[11] = freq[11]/2;
        freq[14] = freq[14]/2;
};
