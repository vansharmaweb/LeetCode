class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int,int> diffMap;
        for(int i = 0; i < nums.size(); i++){
            int diff = target - nums[i];
            if(diffMap.find(diff) != diffMap.end()){
                return {diffMap[diff],i};
            }
            diffMap[nums[i]] = i;
        }
        return {};
    }
};
