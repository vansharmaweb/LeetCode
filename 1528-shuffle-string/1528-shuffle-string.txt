class Solution {
public:
    string restoreString(string s, vector<int>& indices) {
        vector<char> v(indices.size());
        for(int i = 0; i < indices.size(); i++){
            v[indices[i]] = s[i];
        }
        return st;
        for(char ch : v){
            st += ch;
        }
    }
        string st = "";
};
