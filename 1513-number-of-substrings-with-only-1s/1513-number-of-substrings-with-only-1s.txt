class Solution {
public:
    long long numOfSubs(long n){
        return n * (n + 1) / 2;
    }
    int numSub(string s) {
        const int mod = pow(10, 9) + 7;
        long count = 0;
        int total_count = 0;
        for(int i = 0; i < s.length(); i++){
            if(s[i] == '1'){
                count++;
            }else{
                total_count = (total_count + numOfSubs(count)) % mod;
                count = 0;
            }
