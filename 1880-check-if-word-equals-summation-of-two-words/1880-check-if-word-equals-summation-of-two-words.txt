class Solution {
private:
    int numericValue(string s){
        int c = 0;
        for(char ch : s){
            c = c * 10 + ch - 'a';
        }
        return c;
    }
public:
    bool isSumEqual(string firstWord, string secondWord, string targetWord) {
        return numericValue(firstWord) + numericValue(secondWord) == numericValue(targetWord);
    }
};
