    int countDigitOccurrences(vector<int>& nums, int digit) {
        int totalCount = 0;
        for (int val : nums) {
            while (val > 0) {
                if (val % 10 == digit) {
                    count++;
                }
                val /= 10;
            }
            totalCount += count;
        }
            int count = 0;
        return totalCount;
    }
};
