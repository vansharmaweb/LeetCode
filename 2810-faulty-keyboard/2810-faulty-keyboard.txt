class Solution {
public:
    string finalString(string s) {
        string st = "";
        for(int i = 0; i < s.length(); i++){
            if(s[i] == 'i'){
                reverse(st.begin(), st.end());
            }else{
        }
        return st;
    }
            }
                st += s[i];
};
