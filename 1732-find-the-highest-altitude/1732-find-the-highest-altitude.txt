 * Solved on: 2026-10-02

 */

class Solution {
public:
    int largestAltitude(vector<int>& gain) {
        int ans = 0;
        int maxCount = 0;
        for(int val : gain){
            ans += val;
            maxCount = max(maxCount, ans);
        }
        return maxCount;
    }
};
