class Solution {
private:
    bool bOrw(string s){
        int a = s[0];
        int b = s[1];
        int c = a + b;
        if(c % 2 == 0){
            return false;
        }else{
            return true;
        }
    }

public:
    bool checkTwoChessboards(string coordinate1, string coordinate2) {
        return bOrw(coordinate1) == bOrw(coordinate2);
    }
