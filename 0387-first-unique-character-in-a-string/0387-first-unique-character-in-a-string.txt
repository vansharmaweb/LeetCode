class Solution {
public:
    int firstUniqChar(string s) {
        vector<int> chk(26);
        for(char ch : s){
            chk[ch - 'a']++;
        }
        for(int i = 0; i < s.length(); i++){
            if(chk[s[i] - 'a'] == 1){
                return i;
            }
        }
        return -1;
    }
};
