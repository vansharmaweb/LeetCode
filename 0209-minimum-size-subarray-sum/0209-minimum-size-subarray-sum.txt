class Solution {
public:
    int minSubArrayLen(int target, 
    vector<int>& nums) {
        int start = 0;
        int minLength = INT_MAX;
        int winSum = 0;
        for(int end = 0; end < nums.size(); 
        end++){
            winSum += nums[end];
            while(winSum >= target){
                minLength = min(minLength, 
                end - start + 1);
                winSum -= nums[start];
                start++;
            }
