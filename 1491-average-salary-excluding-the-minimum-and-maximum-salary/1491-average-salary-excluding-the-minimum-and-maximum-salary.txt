class Solution {
public:
    double average(vector<int>& salary) {
        sort(salary.begin(), salary.end());
        double sum = 0;
        for(int i = 1; i < n - 1; i++){
            sum += salary[i];
        }
    }
        int n = salary.size();
        return sum/(n-2);
};
