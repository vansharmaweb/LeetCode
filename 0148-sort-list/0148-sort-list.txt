class Solution {
public:
    ListNode* sortList(ListNode* head) {
        vector<int> values;
        ListNode* temp = head;
        while(temp != NULL){
            values.push_back(temp->val);
            temp = temp->next;
        }
        for(int val : values){
            temp->val = val;
        temp = head;
        }
            temp = temp->next;
        return head;
        sort(values.begin(), values.end());
