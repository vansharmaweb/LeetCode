class Solution {
public:
    string interpret(string command) {
        string st = "";
        for(int i = 0; i < command.size(); i++){
            if(command[i] == 'G'){
                st += "G";
            }
            if(command[i]=='(' && command[i+1]==')'){
                st += "o";
                i++;
            }
            if(command[i]=='(' && command[i+1]=='a'){
                st += "al";
                i += 3;
            }
