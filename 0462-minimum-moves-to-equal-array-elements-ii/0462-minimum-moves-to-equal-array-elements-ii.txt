    int minMoves2(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        int diff = 0;
        for(int i = 0; i < n; i++){
            diff = diff + abs(nums[i] - med);
        }
        return diff;
    }
public:
        int med = nums[n/2];
};
