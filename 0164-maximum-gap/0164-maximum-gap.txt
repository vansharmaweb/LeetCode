class Solution {
public:
    int maximumGap(vector<int>& nums) {
        int i = 0;
        int j = 1;
        int ans = 0;
        sort(nums.begin(), nums.end());
        while(j < nums.size()){
            ans = max(ans, (nums[j] - nums[i]));
            i++;
            j++;
        }
        return ans;
    }
};
