class Solution {
public:
    bool containsDuplicate(vector<int>& nums) {
        unordered_map<int, int> mp;
        for(int val : nums){
            if(mp.find(val) != mp.end()){
                return true;
            }else{
        }
                mp[val]++;
            }
    }
        return false;
};
